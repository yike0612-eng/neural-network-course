下面函数的工作是处理**一个**原始分钟数。先预测每个输入会发生什么，再运行或补全函数。

```python
def clean_minutes(raw_value):
    text = str(raw_value).strip().replace("分钟", "")
    if text == "":
        return None, "缺失值"
    if not text.isdigit():
        return None, "无法转为数字"

    minutes = int(text)
    if minutes < 0:
        return None, "不应为负数"
    return minutes, None
```

```python
def summarize(clean_records):
    if len(clean_records) == 0:
        return {"count": 0, "total": 0, "average": 0, "max": None, "min": None}

    total = 0
    values = []
    for record in clean_records:
        total += record["minutes"]
        values.append(record["minutes"])

    return {
        "count": len(clean_records),
        "total": total,
        "average": total / len(clean_records),
        "max": max(values),
        "min": min(values),
    }
```
